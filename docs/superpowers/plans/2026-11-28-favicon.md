- [ ] **Step 1: Write the failing e2e test**

Append to `e2e/home.spec.ts` after the test ending at line 468:

```ts
test("uses the SEC logo as the browser tab icon", async ({ page }) => {
  await page.goto("/", { waitUntil: "networkidle" });

  const icons = page.locator('link[rel~="icon"]');
  await expect(icons).toHaveCount(1);
  await expect(icons.first()).toHaveAttribute("href", /\/favicon\.ico/);

  const icon = await icons.first().evaluate(async (element) => {
    const image = new Image();
    image.src = (element as HTMLLinkElement).href;
    await image.decode();
    const canvas = document.createElement("canvas");
    canvas.width = image.naturalWidth;
    canvas.height = image.naturalHeight;
    canvas.getContext("2d")!.drawImage(image, 0, 0);
    const { data } = canvas.getContext("2d")!.getImageData(0, 0, canvas.width, canvas.height);
    let blue = 0;
    for (let i = 0; i < data.length; i += 4) {
      if (data[i + 3] > 100 && data[i + 2] > data[i] && data[i + 2] > data[i + 1]) blue += 1;
    }
    return { width: canvas.width, height: canvas.height, blue };
  });

  expect(icon.width).toBe(icon.height);
  expect(icon.blue).toBeGreaterThan(20);
});
```

- [ ] **Step 3: Regenerate `src/app/favicon.ico`**

Run from the repository root:

```bash
python3 - <<'PY'
from PIL import Image

source = Image.open("public/sec_logo.png").convert("RGBA")
canvas = Image.new("RGBA", (64, 64), (0, 0, 0, 0))
logo = source.copy()
logo.thumbnail((60, 30), Image.Resampling.LANCZOS)
canvas.alpha_composite(logo, ((64 - logo.width) // 2, (64 - logo.height) // 2))
canvas.save("src/app/favicon.ico", format="ICO", sizes=[(16, 16), (32, 32), (48, 48)])
PY
```

This pads the transparent logo into a square, keeps the source asset untouched, and writes 16/32/48px frames.
